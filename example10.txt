# ══ crashDot — "Neon Drive" ═════════════════════════════════════════════════
# 80s / synthwave / Daft-Punk, built around the new *synthbass* — detuned saws +
# sub through a Moog ladder filter with a snappy filter envelope and drive.
# Load the kit (♪ load kit), then put the cursor on #@intro and Ctrl+Enter — it
# auto-plays and advances. Ctrl+;  (or shutup()) stops everything.
#
# synthbass controls:  sus = note length · detune = saw spread (%) · cutoff/res/fenv
# = the filter + its envelope · sub = weight · drive = warmth · glide = portamento.

#@#@ neon_drive
Clock.bpm = 118
Root.default = "F"
Scale.default = "minor"

#@intro(16)
# the bassline alone — hypnotic root/fifth, filter env pumping, four-on-the-floor
b1 >> synthbass([0, 0, 0, 0, 3, 3, 5, 5], oct=2, dur=1/4, sus=0.22, detune=0.4, cutoff=700, res=0.4, fenv=4, drive=1.6)
d1 >> play("x", dur=1/2, amp=1)

#@groove(24)
# the groove opens up — moving bassline w/ a slow filter, clap on 2&4, offbeat hats
b1 >> synthbass([0, 0, 7, 0, 0, 3, 5, 3], oct=2, dur=1/4, sus=0.2, detune=0.45, cutoff=sinvar([500, 1400], [8]), res=0.42, fenv=4, drive=1.7, pumper=0.7)
d1 >> play("x", dur=1/2, amp=1)
c1 >> play("....o.......o...", dur=1/4, amp=0.9)
h1 >> play("-.-.-.-.-.-.-.-.", dur=1/4, amp=Pacc("offbeat"), hpf=7000)
ld >> synthbass(arp([0, 3, 7, 10], "up"), oct=4, dur=1/8, sus=0.12, detune=0.2, cutoff=2200, res=0.3, fenv=2, drive=1.2, echo=0.25, echo_time=0.375, amp=0.45).every(8, "rotate")

#@lift(16)
# a pad swells in and the filter opens — the synthwave shimmer
p1 >> darkpad(PRoman("i VI III VII"), oct=4, dur=4, sus=4, amp=0.4, cutoff=linvar([600, 3000], [16]), shimmer=0.5)
b1 >> synthbass([0, 0, 7, 0, 0, 3, 5, 3], oct=2, dur=1/4, sus=0.2, detune=0.45, cutoff=linvar([500, 1800], [16]), res=0.42, fenv=4, drive=1.7, pumper=0.7)
d1 >> play("x", dur=1/2, amp=1)
c1 >> play("....o.......o...", dur=1/4, amp=0.9)
h1 >> play("-x-x-x-x-x-x-x-x", dur=1/4, amp=Pacc("offbeat"), hpf=7000)

#@drop(24)
# full drive — bass in front with glide + an octave stab (<3 10>), wide arp, pumping
b1 >> synthbass([0, 0, 7, 0, 0, 3, 5, <3 10>], oct=2, dur=1/4, sus=0.2, detune=0.5, cutoff=1200, res=0.45, fenv=5, drive=2, glide=0.02, pumper=0.75, stereowidth=0.4)
ld >> synthbass(arp([0, 3, 7, 10, 12], "updown"), oct=4, dur=1/8, sus=0.12, detune=0.25, cutoff=sinvar([1200, 4000], [4]), res=0.35, fenv=2, drive=1.3, echo=0.3, echo_time=0.375, pan=<-0.4 0.4>, amp=0.45).every(8, "shuffle")
p1 >> darkpad((0,3,7), oct=4, dur=8, sus=8, amp=0.34, cutoff=2600, shimmer=0.6, room2=0.4)
d1 >> play("x", dur=1/2, amp=1)
c1 >> play("....o.......o...", dur=1/4, amp=0.9)
h1 >> play("-x-x-x-x-x-x-x-x", dur=1/4, amp=Pacc("offbeat"), hpf=7000)
v1 >> play("........X.......", dur=1/4, amp=1)

#@break(16)
# strip to bass + pad, filter closes, big space, long gliding notes with _ rests
b1 >> synthbass([0, _, _, 3, _, _, 5, _], oct=2, dur=1/4, sus=0.6, detune=0.4, cutoff=linvar([1600, 400], [16]), res=0.4, fenv=2, drive=1.4, glide=0.05)
p1 >> darkpad((0,3,7,10), oct=4, dur=8, sus=8, amp=0.4, cutoff=1800, shimmer=0.7, room2=0.6, mix2=0.4)

#@outro(16)
# the bassline fades out under the closing filter
b1 >> synthbass([0, 0, 7, 0, 0, 3, 5, 3], oct=2, dur=1/4, sus=0.2, detune=0.45, cutoff=linvar([1200, 300], [16]), res=0.4, fenv=3, drive=1.6, amp=linvar([0.9, 0], [16]))
d1 >> play("x", dur=1/2, amp=linvar([1, 0], [16]))

#@end(8)

# crashDot — example5 · more codeBank adaptations
# industrial · disco-punk · rave  (machines / disco punk / two unlimited)
# ------------------------------------------------------------------
# CrashServer-only synths/fx mapped to crashDot equivalents (jbass→ebass,
# war→dbass reese, angst→a_gesa, lbass→acidbass), keeping multicrush /
# resonbank / csweep / PFr / drcomp which crashDot has.
# Boot + load the kit. Evaluate a few lines at a time. Scratch pad.


# ═══════════════════════════════════════════════════════════════════════════
#  1 · MACHINES  —  slow industrial, fuzzed bass, swept atmosphere  @ 106
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 106
Root.default = "D"
Scale.default = "minor"

# a fuzzed, multicrushed bass line (jbass → ebass), filter breathing
ba >> ebass([0, 0, -5, 0, -7, 0, -5, -3], oct=3, dur=0.25, sus=var([0.3, 0.2, 0.35, 0.25], [4]), lpf=sinvar([400, 2000], [16]), multicrush=0.8, mclowdrive=1.5, mcmiddrive=2, dist2=0.4, amp=0.7)

# a reese lead (war → dbass), unison, feedback-delayed
~wr >> dbass([0, 0, -5, -5, -7, -7, 0, 0], oct=5, dur=0.5, drive=5, tanh=0.5, lpf=sinvar([600, 3000], [16]), fbdelay=0.5, fbtime=0.25, fbfeed=0.4, fbcutoff=3000, amp=0.4).unison(3)

# a swept, spinning atmosphere (angst → a_gesa)
ag >> a_gesa([7, 5, 0, 7, ., 5, 7, .], oct=4, dur=0.5, distortion=4, cutoff=sinvar([800, 6000], [4]), spin=0.5, fbdelay=0.5, fbtime=0.25, fbfeed=0.7, fbcutoff=3000, amp=0.35)

# a compressed brass stab + a hard kick
br >> brass([0, -5, -7, -5, 0, ., 0, .], oct=5, dur=0.5, sus=0.2, room=0.3, reverb=0.25, comp=0.5, amp=0.4)
k1 >> compkick([0], oct=3, dur=1, punch=6, amp=0.9)


# ═══════════════════════════════════════════════════════════════════════════
#  2 · DISCO PUNK  —  fast, gated tuned kick, acid bass  @ 192
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 192
Root.default = "A"
Scale.default = "minor"

# a driving, gated, fractal-filtered tuned kick
b5 >> compkick([0, 4, 5, (2, 7)], oct=4, dur=0.5, lpf=PFr(1900, 4000), punch=6, rgate=0.5, rgaterate=4, amp=0.8).unison(2)

# a punchy acid bass, fractal filter, chorused
e5 >> acidbass(var([0, -2, -4, -3], 8), oct=4, dur=0.5, lpf=PFr(1900, 6200), lpf_rq=0.15, dist2=0.4, chorus=0.4, amp=0.5)

# punk drums — nested random hats through a bus compressor + a backbeat
x5 >> play("<k.>{-.}[--]{ c}", drcomp=0.5, comp=0.4, hpf=1200, amp=0.6)
a5 >> play("..o.", amp=0.8, echo=0.4, echo_time=0.375)


# ═══════════════════════════════════════════════════════════════════════════
#  3 · TWO UNLIMITED  —  90s rave, resonbank stabs, big hoover  @ 133
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 133
Root.default = "E"
Scale.default = "minor"

# a rave stab — a resonator-bank + cutoff-sweep hit, feedback-delayed
p8 >> play("x.", resonbank=0.8, rbfreq=var([71, 160], [24, 8]), rbdecay=0.4, rbspread=4, csweep=0.4, cswfreq=74, cswrate=0.1, fbdelay=0.5, fbtime=0.25, fbfeed=0.3, fbcutoff=3000, amp=0.6).unison(3)

# a big hoover rave lead
p1 >> hoover([0, (0,3,7), 5, 7], oct=4, dur=0.5, dist2=0.5, lpf=sinvar([600, 4000], [8]), amp=0.4).every(8, "reverse")

# a driving kick, an offbeat Daft filter bass, and offbeat hats
k1 >> play("x", dur=1, amp=0.9)
b1 >> a_daft([0, 0, 5, 3], oct=4, dur=0.5, cutoff=sinvar([300, 2500], [4]), punch=1.5, amp=0.5).offbeat()
h1 >> play("-.-.-.-.", hpf=8000, amp=Pacc("offbeat"))

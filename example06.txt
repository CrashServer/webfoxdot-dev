# crashDot — example06 · more codeBank adaptations
# eurodance · dark synth-pop · dark downtempo
# (EuroDance / Dark Synth Pop / darkChill)
# ------------------------------------------------------------------
# CrashServer-only synths mapped to crashDot (donorgan→organ, lbass→acidbass,
# cbass→bass, viola→prophet); missing fx (valad/tape/follow/slide/fx2) dropped or
# swapped for crashDot ones (PFr, chop, dist2, comp, pong…).
# Boot + load the kit. Evaluate a few lines at a time. Scratch pad.


# ═══════════════════════════════════════════════════════════════════════════
#  1 · EURODANCE  —  rave arps + supersaw hoover  @ 140
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 140
Root.default = "D"
Scale.default = "minor"

# a chopped rave pluck arp on the circle of fifths, fractal filter
p1 >> pluck(PCircle(8), oct=5, dur=PDur(5, 8), sus=0.4, chop=4, lpf=PFr(400, 5000), hpf=200, amp=0.3).unison(3)

# a detuned rave bass stab
j3 >> dbass(var([0, [2, -2]], 16), oct=4, dur=0.5, drive=3, tanh=0.4, lpf=sinvar([500, 3000], [8]), amp=0.5).unison(3)

# a big supersaw hoover lead, flipping every 8 beats
h4 >> supersaw([0, 3, 5, 0, 3, 5, 7, 0], oct=5, dur=0.5, cutoff=linvar([800, 4500], [8]), amp=0.32).every(8, "reverse")

# kick + rave hats + a stuttering clap
q2 >> play("x", dur=1, amp=0.9)
s1 >> play("-.-.-.-.", hpf=8000, amp=Pacc("offbeat"))
s2 >> play("....o...", dur=0.5, room=0.4, amp=0.7).sometimes("stutter", 2)


# ═══════════════════════════════════════════════════════════════════════════
#  2 · DARK SYNTH POP  —  moody organ, acid bass, rhodes  @ 112
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 112
Root.default = "F"
Scale.default = "dorian"

# a moody organ chord riff (donorgan → organ), ping-pong'd and slightly distorted
r8 >> organ(var([6, 5, 3, 1], [8, 4, 2, 2]), oct=5, dur=PDur(3, 8), sus=1, mverb=0.3, dist2=0.4, pong=0.3, pongtime=0.375, amp=0.32).unison(2)

# a driving acid bass (lbass → acidbass), fractal filter
e2 >> acidbass(var([0, 5, 6], [8, 4, 4]), oct=4, dur=0.5, lpf=PFr(1400, 4000, 512), lpf_rq=0.2, chorus=0.4, amp=0.5)

# a warm rhodes counter-melody, sparse
p3 >> rhodes([0, 4, 7, 5], oct=5, dur=2, cutoff=2200, echo=0.4, echo_time=0.375, comp=0.4, amp=0.28)

# 80s drums — a gated snare pattern + offbeat hats
o5 >> play("<x-><..o.>", comp=0.4, amp=0.8)
h1 >> play(".-", dur=0.5, hpf=7000, amp=0.4)


# ═══════════════════════════════════════════════════════════════════════════
#  3 · DARKCHILL  —  deep sub, viola pad, glitch perc  @ 92
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 92
Root.default = "A"
Scale.default = "minor"

# a deep detuned sub, slow filter breath
m0 >> bass(var([0, -2, -4], [32]), oct=3, dur=8, lpf=sinvar([180, 500], [16]), tanh=0.15, amp=0.6).unison(2)

# a viola-like sustained lead (viola → prophet), reverbed, wandering
o9 >> prophet([6, 3, PRand([4, 2, 5])], oct=5, dur=PRand([2, 4, 8]), sus=3, mverb=0.8, lpf=PRand([1200, 3000]), hpf=300, amp=0.4).unison(2) + (-7, 0)

# glitchy downtempo perc — random rate/pan, stutter bursts
t0 >> play("d", dur=0.5, rate=PWhite(1, 3), pan=PWhite(-1, 1), mverb=0.2, amp=Pacc("ghost")).often("stutter", PRand([2, 4, 8]))

# a soft, shaped kick pulse
d6 >> play("x..[x.]x.", dur=0.5, shape=0.4, drcomp=0.4, amp=0.7)

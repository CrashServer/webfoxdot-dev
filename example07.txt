# crashDot — example07 · more codeBank adaptations
# synthwave/acid · doom-metal · ambient drone
# (GTA_2025 / doom / cloudy)
# ------------------------------------------------------------------
# CrashServer-only synths mapped to crashDot (mpluck→moogpluck, tb304→tb303,
# lbass→acidbass); clouds/valad/etc dropped for cs80 + mverb/chorus/spin.
# Boot + load the kit. Evaluate a few lines at a time. Scratch pad.


# ═══════════════════════════════════════════════════════════════════════════
#  1 · GTA  —  synthwave + acid, cross-player harmony  @ 164
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 164
Root.default = "C"
Scale.default = "dorian"

# a moogpluck arp motif, reverb blooming in over 32 beats, panning
t1 >> moogpluck([0, 1, 2, 1, 0, 1, 0, -1], oct=5, dur=2, mverb=lininf(0, 0.8, 32), pan=PWhite(-1, 1), hpf=400, amp=0.35).unison(4)

# a tb303 acid line tracing the same motif
t2 >> tb303([0, 1, 2, 1, 0, 1, 0, -1], oct=6, dur=0.5, lpf=linvar([300, 4000], [24]), lpf_rq=0.15, hpf=500, mverb=0.4, amp=0.35).unison(3).sometimes("stutter", 2)

# a second acid a fifth above t2, tracked live
t3 >> tb303(t2.degree + 5, oct=5, dur=0.5, lpf=sinvar([600, 3500], [8]), lpf_rq=0.2, amp=0.3)

# a wide detuned ssaw pad
t5 >> ssaw([0, 2, (3,4), (6,7)], oct=5, dur=8, lpf=7000, chorus=0.5, amp=0.35).unison(4)

# a driving kick + offbeat hats
k1 >> play("x", dur=1, amp=0.8)
h1 >> play("-.-.-.-.", hpf=8000, amp=Pacc("offbeat"))


# ═══════════════════════════════════════════════════════════════════════════
#  2 · DOOM  —  distorted guitar-tuned riff, screaming acid  @ 170
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 170
Root.default = "E"
Scale.default = "minor"

# a distorted, palm-muted guitar-tuned riff (faim .gtr)
g1 >> faim([-5, -5, 7, -5, -5, 5, -5, -5, 3, -5, -5, 1], oct=4, dur=0.5, dist2=1, shape=0.3, lpf=PRand([600, 2200]), amp=0.4).gtr(2)

# a screaming acid — twelve-tone-ish rows swapped each phrase (Pvar), gtr-tuned
g2 >> tb303(Pvar([[9, 7, 6, 9, 12, 10, 9, 6], [14, 10, 7, 10, 14, 10, 14, 19]], 32), oct=4, dur=0.25, dist2=0.5, lpf=linvar([400, 5000], [8]), lpf_rq=0.2, amp=0.35).gtr(2)

# a low reese bass (lbass → acidbass)
g3 >> acidbass(var([0, [3, 5, 7, 0]], [14, 2]), oct=4, dur=0.5, lpf=1700, lpf_rq=0.2, dist2=0.4, amp=0.5).gtr(1)

# blast-beat drums + a hard electro kick
g4 >> play("x[oo]x{o }x[xx]o{ c}", dur=0.25, drcomp=0.5, comp=0.4, amp=0.7).human(40, 0).every(14, "stutter", 8)
k1 >> a_bd([0], oct=2, dur=0.5, distortion=5, punch=7, amp=0.9)


# ═══════════════════════════════════════════════════════════════════════════
#  3 · CLOUDY  —  evolving cs80 pad drone, ambient  @ 122
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 122
Root.default = "A"
Scale.default = "minor"

# an evolving cs80 pad drone, filter sweeping slowly on a huge reverb
c3 >> cs80((0, 3, 7), oct=3, dur=8, sus=6, lpf=linvar([400, 4000], [28, 4]), mverb=0.85, mverbmix=0.7, amp=0.4)

# a shifting cs80 texture — filter jumps, vibrato, spinning
c1 >> cs80([0, 4, 7, 11], oct=4, dur=0.5, cutoff=linvar([400, 4000, 400], [8, 8]), chorus=0.5, vibrato=0.5, vib_rate=3, spin=0.4, amp=0.28)

# a distant bell motif on an arch contour, sometimes reversed
b3 >> bell(PContour("arch", 6, 7), oct=6, dur=4, mverb=0.9, echo=0.4, echo_time=0.5, amp=0.2).sometimes("mirror")

# a soft sub pulse underneath
b1 >> dbass([0, 0, 5, 3], oct=3, dur=4, tanh=0.15, lpf=500, amp=0.4)

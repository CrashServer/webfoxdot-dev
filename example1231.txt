# ════════════════════════════════════════════════════════════════════════════
#  example1231 — MINIMALISM:  Philip Glass additive + Steve Reich phasing
#
#  Two classic techniques, both already possible with what's here:
#   • Reich "phasing" (dephasing): two IDENTICAL cells, one drifting a hair
#     slower (dur * 1.01…), so they slide in and out of alignment. Pan them
#     L/R and you HEAR the phase move. No special function needed — just dur.
#   • Glass "additive process": an arpeggio that GROWS a note at a time —
#     PGrowArp([0,2,4,7]) → 0 · 0 2 · 0 2 4 · 0 2 4 7 …  (this IS the tool).
#
#  The only thing worth adding is a bright music-box timbre — so there's a
#  `celeste` defined inline below with defsynth() (no engine rebuild).
#
#  HOW TO PLAY:  run the defsynth block ONCE (cursor on it, Ctrl+Alt+Enter),
#  then run the parts. Give it a minute — the whole point is the slow drift.
# ════════════════════════════════════════════════════════════════════════════

# ── the timbre — a bright music-box / celeste  (run this block ONCE) ──────────
defsynth("celeste", { bright: 1 }, ({ out, note, amp, sus, pan, attack, release, bright }) => {
  const f = note.midicps()
  const env = EnvGen.ar(Env.perc(0.001, sus, 1, -4), { doneAction: 2 })
  let sig = SinOsc.ar(f)
  sig = sig.add(SinOsc.ar(f.mul(2)).mul(0.5))
  sig = sig.add(SinOsc.ar(f.mul(3)).mul(0.25).mul(bright))
  sig = sig.add(SinOsc.ar(f.mul(4.2)).mul(0.12).mul(bright))   // slight inharmonic shimmer
  sig = sig.mul(env).mul(amp).mul(0.4)
  Out.ar(out, Pan2.ar(sig, pan))
})


Clock.bpm = 138
Scale.default = "major"
Root.default = "C"


# ── PART 1 — REICH PHASING ────────────────────────────────────────────────────
# The SAME 6-note cell, twice. p2's dur is 1.2% longer, so it drifts; panned hard
# L/R you hear the two lines walk apart and slowly come back. This is the piece.
p1 >> celeste([0, 2, 4, 7, 4, 2], oct=5, dur=1/6,         sus=0.4, amp=0.4, pan=-0.4, room=0.6, reverb=0.35)
p2 >> celeste([0, 2, 4, 7, 4, 2], oct=5, dur=1/6 * 1.012, sus=0.4, amp=0.4, pan=0.4,  room=0.6, reverb=0.35)


# ── PART 2 — GLASS ADDITIVE PROCESS ──────────────────────────────────────────
# The arpeggio grows a note at a time and loops — hypnotic, always the same yet
# always moving. Different subdivision (1/8) from the phasing cells (1/6) → a
# shifting polyrhythm against them.
ar >> organ(PGrowArp([0, 2, 4, 7, 9]), oct=5, dur=1/8, sus=0.12, amp=0.24, lpf=4500, room=0.4)


# ── PART 3 — SLOW HARMONY ─────────────────────────────────────────────────────
# A chord bed and bass moving on a gentle progression (I V vi IV), one chord
# every 2 bars — the slow harmonic rhythm that lets the fast figures float.
pd >> darkpad(PProg("pop"), oct=4, dur=8, sus=8, amp=0.28, attack=2.5, cutoff=1400, reverb=0.6, room=0.9, chorus=0.4)
bs >> sine([0, 4, 5, 3], oct=2, dur=8, sus=6, amp=0.5, lpf=700)      # roots of I V vi IV

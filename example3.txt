# crashDot — example3 · NERVOUS + AGGRESSIVE
# breakcore · gabber · glitch — jittery drums, distorted lines
# (inspired by codeBank: annihilation, postGabber, glitch)
# ------------------------------------------------------------------
# Boot + load the kit. Evaluate a few lines at a time. Ctrl+; stops all.
# A scratch pad — NOT wired into the in-app examples.


# ═══════════════════════════════════════════════════════════════════════════
#  1 · ANNIHILATION  —  breakcore · distorted reese · screaming acid  @ 150
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 150
Root.default = "E"
Scale.default = "phrygian"

# nervous glitch drums — nested random/subdivided steps, random dur/rate/pan/filter
d1 >> play("{x x}[--]x.{o-}[xx].{c }x", dur=PRand([1/4, 1/2]), lpf=PWhite(800, 9000), rate=PwRand([1, 2, -1], [5, 2, 1]), pan=PWhite(-0.8, 0.8), amp=Pacc("ghost")).sometimes("stutter", 4)
# a driving distorted kick underneath
k1 >> play("x...x...x...x...", amp=0.9, drive=3, tanh=0.4)

# a distorted reese bass — gated, fractal filter, mean
b1 >> ebass(var([0, 3, 5], 8), oct=3, dur=0.25, dist2=0.7, dist2shape=1, lpf=PFr(300, 4000), rgate=0.7, rgaterate=8, amp=0.5)

# a screaming acid line — resonant, crushed, sweeping, flipped every 8 beats
p1 >> tb303([0, 0, 3, 0, 7, 3, 5, 0], oct=3, dur=0.25, lpf=linvar([300, 5000], [8]), lpf_rq=0.1, dist2=0.5, crush=0.4, bits=6, amp=0.35).every(8, "reverse")

# noise stabs — pitched hat, random bursts on the offbeat
h1 >> a_hhat([0], oct=6, dur=PRand([1/4, 1/8]), tone=PWhite(3000, 9000), distortion=2, amp=Pacc("offbeat")).sometimes("stutter", 3)


# ═══════════════════════════════════════════════════════════════════════════
#  2 · POSTGABBER  —  hardcore · distorted kick · wobble + supersaw  @ 175
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 175
Root.default = "F"
Scale.default = "minor"

# gabber kick — a_bd driven hard, doubled with a dbass overdrive layer
k1 >> a_bd([0], oct=2, dur=1, distortion=6, punch=8, amp=1)
k2 >> dbass([0], oct=2, dur=1, drive=8, tanh=0.8, shape=0.7, crush=0.5, bits=4, amp=0.6)

# jittery hats + a rushing snare roll
h1 >> play("-[--]-[--]", dur=0.25, hpf=8000, amp=Pacc("ghost"))
d1 >> play(".{o }.{o[oo]}", dur=0.5, amp=0.7).sometimes("stutter", 6)

# a wobble bass talking under it
b1 >> wobble([0, 0, 3, 5], oct=3, dur=1, rate=8, cutoff=7000, dist2=0.4, amp=0.6)

# an aggressive supersaw rave stab, filter pumping, on the offbeat
p1 >> supersaw([0, (0,3,7)], oct=4, dur=0.5, cutoff=sinvar([500, 4500], [4]), drive=4, tanh=0.5, amp=0.35).offbeat()


# ═══════════════════════════════════════════════════════════════════════════
#  3 · GLITCH  —  IDM · chaotic modulation · stutter everything  @ 160
# ═══════════════════════════════════════════════════════════════════════════
Clock.bpm = 160
Root.default = "A"
Scale.default = "minPentatonic"

# chaotic drums — Euclidean grid, crushed, chaos-driven filter, random flips
d1 >> play(PEuclid2(7, 16, ".", "x"), dur=0.25, crush=0.6, bits=PRand([3, 4, 6]), lpf=PLorenz(500, 6000), amp=0.7).sometimes("stutter", 8)
d2 >> play("{c-}{Xo}[cc]{ x}", dur=PRand([1/4, 1/8]), pan=PBrown(-1, 1), rate=PwRand([1, 2, 4, -1], [6, 3, 1, 1]), amp=Pacc("ghost"))

# a glitchy lead — heavy crush/shape, no-repeat random pitch, chaotic band-pass
p1 >> blip(PxRand(0, 12), oct=6, dur=PRand([1/4, 1/8]), crush=0.5, bits=4, shape=0.6, bpf=PLorenz(600, 4000), amp=0.3).every(4, "shuffle")

# a chaotic sub drifting on a Lorenz / Brownian attractor
b1 >> a_gesa([0, 0, -2, 3], oct=2, dur=0.5, distortion=6, cutoff=PBrown(400, 3000), tanh=0.4, amp=0.6)
